cmake_minimum_required(VERSION 3.22)
project(Ultimate VERSION 0.1.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(Ultimate COMPANY_NAME "Independent Audio" BUNDLE_ID "audio.independent.ultimate.v01" PLUGIN_MANUFACTURER_CODE InAu PLUGIN_CODE Ul01 FORMATS VST3 PRODUCT_NAME "Ultimate Channel Strip")
juce_generate_juce_header(Ultimate)
target_sources(Ultimate PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h)
target_compile_definitions(Ultimate PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Ultimate PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_processors juce::juce_audio_plugin_client)
